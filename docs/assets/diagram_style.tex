% Shared TikZ styles for the box diagrams in assets/.
% Every box has the same width (\diagboxwidth) and minimum height, so
% columns line up (break lines with \\; no automatic wrapping); arrows use one tip and line weight throughout.
\usetikzlibrary{arrows.meta,positioning,fit,backgrounds,calc}

\newcommand{\diagboxwidth}{3.4cm}

\tikzset{
	diagram/.style={
			font=\sffamily\small,
			node distance=6mm and 8mm,
			>={Stealth[length=2.4mm,width=1.8mm]},
		},
	block/.style={
			draw=black!60,
			fill=black!3,
			rounded corners=2pt,
			line width=0.7pt,
			minimum width=\diagboxwidth,
			minimum height=11mm,
			inner sep=5pt,
			align=center,
		},
	frozen/.style={block, draw=blue!60!black, fill=blue!6},
	adapter/.style={block, draw=teal!60!black, fill=teal!7},
	result/.style={block, draw=orange!65!black, fill=orange!8},
	plain/.style={block},
	% Circled operator, e.g. the sum of the two paths.
	op/.style={
			circle,
			draw=black!60,
			fill=white,
			line width=0.7pt,
			inner sep=0pt,
			minimum size=7mm,
			font=\sffamily\bfseries\normalsize,
		},
	arrow/.style={->, draw=black!70, line width=0.7pt, rounded corners=3pt},
	frozen-link/.style={arrow, densely dashed},
	feedback/.style={arrow, densely dashed, draw=teal!60!black},
	% Dashed frame grouping the trainable adapter factors.
	group/.style={
			draw=teal!60!black,
			densely dashed,
			rounded corners=4pt,
			line width=0.6pt,
			inner sep=6pt,
		},
	group-label/.style={font=\sffamily\footnotesize\itshape, text=teal!50!black},
	% Panel used when several diagrams share one image.
	panel/.style={
			draw=black!25,
			rounded corners=4pt,
			line width=0.7pt,
			inner sep=10pt,
		},
	panel-title/.style={font=\sffamily\bfseries\normalsize},
}
